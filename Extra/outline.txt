```{r msleeptbl3}
kable(msleep[1:5,1:7], caption="Stylized Kable") %>%
  kable_styling(bootstrap_options = c("striped", "condensed", "hover"),
                full_width = FALSE,
                position="left")%>%
  row_spec(0, background = "#95c5c5", color = "white") %>%
  column_spec(1, bold = TRUE, border_right = "2px solid #555") %>%
  footnote(general = "This is a subset of the `msleep` dataset.",
           footnote_as_chunk = TRUE)
```
